PROCEDURE enforceDriverIdentification(violationList, licenseInput, altIdType, altIdNumber): IF violationList CONTAINS "No Driver's License" THEN // Unlicensed motorist branch SET isUnlicensed ← TRUE DISABLE licenseInput SET licenseInput.value ← "N/A - Unlicensed" SHOW altIdInputGroup VALIDATE altIdType IS NOT EMPTY IF VALIDATION FAILS THEN EMIT ERROR "Alternative ID type is required for unlicensed drivers." END IF VALIDATE altIdNumber IS NOT EMPTY IF VALIDATION FAILS THEN EMIT ERROR "Alternative ID number is required for unlicensed drivers." END IF ELSE // Standard licensed driver branch SET isUnlicensed ← FALSE ENABLE licenseInput HIDE altIdInputGroup VALIDATE licenseInput.value IS NOT EMPTY AND != "N/A - UNLICENSED" IF VALIDATION FAILS THEN EMIT ERROR "Valid Driver's License Number is mandatory." END IF SET altIdType ← NULL, altIdNumber ← NULL END IF

FUNCTION reconcileFineTotal(selectedViolations, fineAmount, fineOverrideFlag): INITIALIZE statutorySum ← 0.00 // Compute exact statutory baseline from Schedule FOR EACH violation IN selectedViolations DO SELECT CASE violation CASE "No Driver's License": statutorySum ← statutorySum + 1500.00 CASE "Expired Registration": statutorySum ← statutorySum + 1200.00 CASE "No Helmet / Seatbelt": statutorySum ← statutorySum + 1000.00 CASE "DUI / Impaired Driving": statutorySum ← statutorySum + 5000.00 CASE "Illegal Modification": statutorySum ← statutorySum + 2500.00 CASE "Reckless Driving": statutorySum ← statutorySum + 3000.00 CASE DEFAULT: statutorySum ← statutorySum + 1000.00 END SELECT END FOR // Enforce non-negative numeric fine VALIDATE fineAmount IS FINITE AND fineAmount > 0.00 IF VALIDATION FAILS THEN EMIT ERROR "Fine amount must be a positive numerical value." END IF // Branch based on discretionary override toggle IF fineOverrideFlag == 1 THEN RETURN { isValid: TRUE, amount: fineAmount, isOverridden: 1 } ELSE IF ABS(fineAmount - statutorySum) > 0.05 THEN EMIT ERROR "Matching fine total required: ₱" + statutorySum RETURN { isValid: FALSE } END IF RETURN { isValid: TRUE, amount: statutorySum, isOverridden: 0 } END IF

PROCEDURE enforceVehicleDisposition(disposition, impoundReceiptInput): // Check if disposition entails impounding at police station IF disposition == "Impounded" OR disposition == "Impounded at Base / Station" THEN SHOW impoundReceiptContainer SET impoundReceiptInput.required ← TRUE // Mandate towing slip serial number VALIDATE impoundReceiptInput.value IS NOT EMPTY IF VALIDATION FAILS THEN EMIT ERROR "Impound Receipt / Towing Slip Number is required when vehicle disposition is Impounded." HALT INGESTION END IF SET storedSlipNo ← TRIM(impoundReceiptInput.value) ELSE // Vehicle released with citation or turned over HIDE impoundReceiptContainer SET impoundReceiptInput.required ← FALSE CLEAR impoundReceiptInput.value SET storedSlipNo ← NULL END IF
